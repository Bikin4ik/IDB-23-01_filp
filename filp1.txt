import tkinter as tk
from tkinter import ttk, messagebox
import datetime
import docx
import os

def main():
    root = tk.Tk()
    root.title("Генератор документов")
    root.geometry("550x450")

    # Переменные для хранения данных
    fio_var = tk.StringVar()
    position_var = tk.StringVar()
    subject_var = tk.StringVar()
    gender_var = tk.StringVar(value="male")  # male или female

    # Функция получения окончаний в зависимости от пола
    def get_gender_suffixes(gender):
        if gender == "male":
            return {
                "occupies": "занимающий",
                "prepared": "подготовил",
            }
        else:
            return {
                "occupies": "занимающая",
                "prepared": "подготовила",

            }

    # Функция генерации документа
    def generate():
        fio = fio_var.get().strip()
        position = position_var.get().strip()
        subject = subject_var.get().strip()
        gender = gender_var.get()

        if not all([fio, position, subject]):
            messagebox.showwarning("Ошибка", "Заполните все поля!")
            return

        try:
            doc = docx.Document()

            # Заголовок
            title = doc.add_heading('СЛУЖЕБНАЯ ЗАПИСКА', 0)
            title.alignment = docx.enum.text.WD_ALIGN_PARAGRAPH.CENTER

            # Дата
            current_date = datetime.datetime.now().strftime("%d.%m.%Y")
            date_para = doc.add_paragraph()
            date_para.alignment = docx.enum.text.WD_ALIGN_PARAGRAPH.RIGHT
            date_para.add_run(f"Дата: {current_date}").bold = True

            doc.add_paragraph()

            # Данные отправителя
            doc.add_heading('Отправитель:', 2)
            doc.add_paragraph(f"ФИО: {fio}\nДолжность: {position}")

            doc.add_paragraph()

            # Тема
            doc.add_heading('Тема:', 2)
            doc.add_paragraph(subject).bold = True

            doc.add_paragraph()

            # Получаем окончания в зависимости от пола
            suffixes = get_gender_suffixes(gender)

            # Основной текст с учетом пола
            doc.add_heading('Содержание:', 2)
            content = f"""Уважаемый руководитель!

Настоящим докладываю, что в соответствии с возложенными на меня обязанностями, я, {fio}, {suffixes['occupies']} должность {position}, {suffixes['prepared']} настоящий документ 
по следующей теме: "{subject}".
Прошу Вас рассмотреть данный документ и принять соответствующее решение.
Дата составления: {current_date}
С уважением,
{fio}
{position}"""
            doc.add_paragraph(content)

            # Подпись
            doc.add_paragraph()
            signature_para = doc.add_paragraph()
            signature_para.alignment = docx.enum.text.WD_ALIGN_PARAGRAPH.RIGHT
            signature_para.add_run("___________________").bold = True
            signature_para.add_run("\n(подпись)")

            # Сохранение
            filename = f"документ_{datetime.datetime.now().strftime('%Y%m%d_%H%M%S')}.docx"
            doc.save(filename)

            messagebox.showinfo("Успех", f"Документ успешно создан!\nСохранен как: {filename}")
            os.startfile(os.path.dirname(os.path.abspath(filename)))

        except Exception as e:
            messagebox.showerror("Ошибка", f"Произошла ошибка:\n{str(e)}")

    # Интерфейс
    frame = ttk.Frame(root, padding="20")
    frame.pack(fill=tk.BOTH, expand=True)

    # Заголовок
    ttk.Label(frame, text="Генератор документов", font=('Arial', 14, 'bold')).pack(pady=(0, 20))

    # Поле ФИО
    ttk.Label(frame, text="ФИО:", font=('Arial', 10)).pack(anchor=tk.W, pady=(0, 5))
    ttk.Entry(frame, textvariable=fio_var, width=45, font=('Arial', 10)).pack(fill=tk.X, pady=(0, 10))

    # Поле Должность
    ttk.Label(frame, text="Должность:", font=('Arial', 10)).pack(anchor=tk.W, pady=(0, 5))
    ttk.Entry(frame, textvariable=position_var, width=45, font=('Arial', 10)).pack(fill=tk.X, pady=(0, 10))

    # Поле Тема письма
    ttk.Label(frame, text="Тема письма:", font=('Arial', 10)).pack(anchor=tk.W, pady=(0, 5))
    ttk.Entry(frame, textvariable=subject_var, width=45, font=('Arial', 10)).pack(fill=tk.X, pady=(0, 15))

    # Выбор пола (галочки - Radio buttons)
    gender_frame = ttk.LabelFrame(frame, text="Пол", padding="10")
    gender_frame.pack(fill=tk.X, pady=(0, 15))

    ttk.Radiobutton(gender_frame, text="Мужской", variable=gender_var, value="male").pack(side=tk.LEFT, padx=(0, 20))
    ttk.Radiobutton(gender_frame, text="Женский", variable=gender_var, value="female").pack(side=tk.LEFT)

    # Кнопка генерации
    generate_btn = ttk.Button(frame, text="Сгенерировать документ", command=generate,
                              style="Accent.TButton")
    generate_btn.pack(pady=10)

    # Информационная метка
    info_label = ttk.Label(frame, text="Нажмите кнопку для создания документа Word\n"
                                       "Документ будет сохранен в текущей папке",
                           font=('Arial', 9), foreground='gray')
    info_label.pack(pady=(15, 0))

    # Настройка стилей
    style = ttk.Style()
    style.configure("Accent.TButton", font=('Arial', 10, 'bold'))

    root.mainloop()

if __name__ == "__main__":
    main()
